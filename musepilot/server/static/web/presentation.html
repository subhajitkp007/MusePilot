class StepNode(BaseModel):
    model_config = ConfigDict(extra="forbid")
    step_id: int = Field(..., ge=1)
    operation: Literal["click", "type", "drag", "wait", "verify"]

    drag_to_mask_id: int | None = None

    @model_validator(mode="before")

        violating = FORBIDDEN_COORDINATE_KEYS.intersection(data.keys())
        if violating:
            raise ZeroCoordinateViolationError(f"Forbidden keys: {violating}")
        return data
